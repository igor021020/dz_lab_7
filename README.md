# #1. Алгоритм
Начало

Ввод исходных данных: номер месяца (month), год (year).

Проверка корректности введенного номера месяца (от 1 до 12).

Проверка года на високосность.

Определение количества дней в месяце в зависимости от порядкового номера и високосности года.

Формирование правильного склонения слова "год" (год / года / лет).

Вывод результата.

Конец

# #2.Реализация программы
```c
#include <stdio.h>
#include <locale.h>

int main() {
    setlocale(LC_ALL, "Russian");

    int month, year;
    int days;
    int is_leap = 0;
    const char* month_names[] = {
        "", "Январь", "Февраль", "Март", "Апрель", "Май", "Июнь",
        "Июль", "Август", "Сентябрь", "Октябрь", "Ноябрь", "Декабрь"
    };

    printf("Введите номер месяца (1-12): ");
    if (scanf("%d", &month) != 1 || month < 1 || month > 12) {
        printf("Ошибка: введите корректный номер месяца (от 1 до 12).\n");
        return 1;
    }

    printf("Введите год: ");
    if (scanf("%d", &year) != 1) {
        printf("Ошибка: введите корректный год.\n");
        return 1;
    }

    if ((year % 4 == 0 && year % 100 != 0) || (year % 400 == 0)) {
        is_leap = 1;
    }

    switch (month) {
    case 1: case 3: case 5: case 7: case 8: case 10: case 12:
        days = 31;
        break;
    case 4: case 6: case 9: case 11:
        days = 30;
        break;
    case 2:
        days = is_leap ? 29 : 28;
        break;
    }

    const char* year_word;
    int last_two = year % 100;
    int last_one = year % 10;

    if (last_two >= 11 && last_two <= 14) {
        year_word = "лет";
    }
    else if (last_one == 1) {
        year_word = "год";
    }
    else if (last_one >= 2 && last_one <= 4) {
        year_word = "года";
    }
    else {
        year_word = "лет";
    }

    printf("\nМесяц: %s\n", month_names[month]);
    printf("Год: %d (%s)\n", year, is_leap ? "високосный" : "не високосный");
    printf("Количество дней: %d\n", days);
    printf("В %s %d %s — %d дней.\n", month_names[month], year, year_word, days);

    return 0;
}
```
